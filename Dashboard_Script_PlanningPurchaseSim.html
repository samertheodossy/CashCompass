/* Planning — Purchase simulator tab */ function togglePurchaseFields() { const mode = document.getElementById('ps_fundingMode').value; document.getElementById('ps_cashDownWrap').style.display = mode === 'Split' ? 'block' : 'none'; document.getElementById('ps_financeWrap').style.display = (mode === 'Finance' || mode === 'Split') ? 'block' : 'none'; } function loadPurchaseSimulatorSection() { const initialLoadTiming = startDashboardInitialLoadStage_('purchase_simulator'); resetPurchaseSimulationResults_(); google.script.run .withSuccessHandler(function(data) { clearSurfaceState_('ps_status', 'status'); const d = data.defaults || {}; document.getElementById('ps_fundingMode').value = d.fundingMode || 'Cash'; document.getElementById('ps_aprPct').value = d.aprPct || 7.5; document.getElementById('ps_termMonths').value = d.termMonths || 60; document.getElementById('ps_monthlyRunningCost').value = ''; document.getElementById('ps_treatAsAsset').checked = !!d.treatAsAsset; togglePurchaseFields(); finishDashboardInitialLoadStage_(initialLoadTiming, 'ok'); }) .withFailureHandler(function(err) { resetPurchaseSimulationResults_(); renderSurfaceState_('ps_status', { kind: 'unavailable', title: 'Purchase simulator is unavailable right now.', message: customerSafeErrorMessage_(err && err.message ? err.message : String(err), 'CashCompass could not load the purchase simulator. Try again in a moment.'), actionLabel: 'Try again', onAction: loadPurchaseSimulatorSection }); finishDashboardInitialLoadStage_(initialLoadTiming, 'error'); }) .getPurchaseSimulatorUiData(); } function resetPurchaseSimulationResults_() { const guidance = document.getElementById('ps_guidance'); const results = document.getElementById('ps_results'); if (guidance) guidance.hidden = false; if (results) results.hidden = true; } function showPurchaseSimulationResults_() { const guidance = document.getElementById('ps_guidance'); const results = document.getElementById('ps_results'); if (guidance) guidance.hidden = true; if (results) results.hidden = false; } function runPurchaseSimulationUi() { const payload = { purchaseName: document.getElementById('ps_name').value, purchaseAmount: toNumber(document.getElementById('ps_amount').value), fundingMode: document.getElementById('ps_fundingMode').value, cashDown: toNumber(document.getElementById('ps_cashDown').value), aprPct: Number(document.getElementById('ps_aprPct').value || 0), termMonths: Number(document.getElementById('ps_termMonths').value || 0), monthlyRunningCost: toNumber(document.getElementById('ps_monthlyRunningCost').value), treatAsAsset: document.getElementById('ps_treatAsAsset').checked }; resetPurchaseSimulationResults_(); setStatusLoading('ps_status', 'Running simulation…'); google.script.run .withSuccessHandler(function(res) { setStatus('ps_status', 'Simulation complete.', false); document.getElementById('ps_immediateCashHit').textContent = fmtCurrency(res.immediateCashHit); document.getElementById('ps_financedAmount').textContent = fmtCurrency(res.financedAmount); document.getElementById('ps_monthlyPayment').textContent = fmtCurrency(res.monthlyPayment); document.getElementById('ps_totalMonthlyImpact').textContent = fmtCurrency(res.totalMonthlyImpact); document.getElementById('ps_newCash').textContent = fmtCurrency(res.newCash); document.getElementById('ps_newUsableCash').textContent = fmtCurrency(res.newUsableCash); document.getElementById('ps_newProjectedCashFlow').textContent = fmtCurrency(res.newProjectedCashFlow); document.getElementById('ps_netWorthChange').textContent = fmtCurrency(res.immediateNetWorthChange); document.getElementById('ps_runwayBefore').textContent = res.runwayBefore ? res.runwayBefore.label : '—'; document.getElementById('ps_runwayAfter').textContent = res.runwayAfter ? res.runwayAfter.label : '—'; document.getElementById('ps_warning').textContent = res.warning || '—'; const nw = document.getElementById('ps_netWorthChange'); nw.className = 'muted'; if (Number(res.immediateNetWorthChange) < 0) nw.className = 'accent-bad'; if (Number(res.immediateNetWorthChange) === 0) nw.className = 'accent-good'; const cf = document.getElementById('ps_newProjectedCashFlow'); cf.className = Number(res.newProjectedCashFlow) < 0 ? 'accent-bad' : 'accent-good'; showPurchaseSimulationResults_(); }) .withFailureHandler(function(err) { resetPurchaseSimulationResults_(); renderSurfaceState_('ps_status', { kind: 'error', title: 'The simulation could not be completed.', message: customerSafeErrorMessage_(err && err.message ? err.message : String(err), 'Review the purchase details, then try the simulation again.') }); }) .runPurchaseSimulation(payload); }
